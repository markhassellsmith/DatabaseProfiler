CREATE OR ALTER PROCEDURE dbo.ProfileTable
(
      @TableName        NVARCHAR(256)   -- Expect schema.table
    , @SamplePercent    FLOAT = NULL    -- NULL = full scan; else 0 < pct <= 100
)
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY

        --------------------------------------------------------------------------
        -- 1. Validate & Parse Schema/Table
        --------------------------------------------------------------------------
        DECLARE @SchemaName SYSNAME, @ObjectName SYSNAME;

        SET @SchemaName = PARSENAME(@TableName, 2);
        SET @ObjectName = PARSENAME(@TableName, 1);

        IF @SchemaName IS NULL OR @ObjectName IS NULL
        BEGIN
            RAISERROR('Input must be schema.table', 16, 1);
            RETURN;
        END

        IF NOT EXISTS (
             SELECT 1 FROM sys.objects o
             JOIN sys.schemas s ON o.schema_id = s.schema_id
             WHERE o.name = @ObjectName
               AND s.name = @SchemaName
               AND o.type = 'U'
        )
        BEGIN
            RAISERROR('Table does not exist', 16, 1);
            RETURN;
        END

        --------------------------------------------------------------------------
        -- 2. Prepare temp table for result set R
        --------------------------------------------------------------------------
        CREATE TABLE #ProfileResults
        (
              ColumnName      SYSNAME
            , DataType        SYSNAME
            , RowCount        BIGINT
            , NullCount       BIGINT
            , DistinctCount   BIGINT
            , MinValue        SQL_VARIANT NULL
            , MaxValue        SQL_VARIANT NULL
            , AvgValue        FLOAT NULL
            , MaxLength       INT NULL
        );

        --------------------------------------------------------------------------
        -- 3. Build FROM clause with optional sampling
        --------------------------------------------------------------------------
        DECLARE @FromClause NVARCHAR(MAX);

        IF @SamplePercent IS NULL OR @SamplePercent <= 0 OR @SamplePercent > 100
        BEGIN
            SET @FromClause = QUOTENAME(@SchemaName) + '.' + QUOTENAME(@ObjectName);
        END
        ELSE
        BEGIN
            -- TABLESAMPLE gives row-based sampling (not perfect distribution but fast)
            SET @FromClause =
                QUOTENAME(@SchemaName) + '.' + QUOTENAME(@ObjectName)
                + ' TABLESAMPLE (' + CAST(@SamplePercent AS NVARCHAR(30)) + ' PERCENT)';
        END

        --------------------------------------------------------------------------
        -- 4. Iterate columns using APPLY-based profiling
        --------------------------------------------------------------------------
        DECLARE @DynamicSQL NVARCHAR(MAX) = N'';

        SELECT @DynamicSQL = STRING_AGG( ProfileSQL, CHAR(10) )
        FROM (
            SELECT
                ProfileSQL =
                '
                INSERT INTO #ProfileResults
                SELECT
                      ''' + c.name + ''' AS ColumnName
                    , ''' + t.name + ''' AS DataType
                    , Stats.RowCount
                    , Stats.NullCount
                    , Stats.DistinctCount
                    , Stats.MinValue
                    , Stats.MaxValue
                    , Stats.AvgValue
                    , Stats.MaxLength
                FROM (
                    SELECT *
                    FROM ' + @FromClause + ' AS T
                    OUTER APPLY (
                        SELECT
                              COUNT(*) AS RowCount
                            , SUM(CASE WHEN T.' + QUOTENAME(c.name) + ' IS NULL THEN 1 ELSE 0 END) AS NullCount
                            , COUNT(DISTINCT T.' + QUOTENAME(c.name) + ') AS DistinctCount

                            -- Conditional MIN
                            , CASE 
                                  WHEN t.name IN (''int'',''bigint'',''smallint'',''tinyint'',''decimal'',''numeric'',''float'',''real'',''money'',''smallmoney'',''date'',''datetime'',''datetime2'',''smalldatetime'',''datetimeoffset'')
                                  THEN MIN(T.' + QUOTENAME(c.name) + ')
                              END AS MinValue

                            -- Conditional MAX
                            , CASE 
                                  WHEN t.name IN (''int'',''bigint'',''smallint'',''tinyint'',''decimal'',''numeric'',''float'',''real'',''money'',''smallmoney'',''date'',''datetime'',''datetime2'',''smalldatetime'',''datetimeoffset'')
                                  THEN MAX(T.' + QUOTENAME(c.name) + ')
                              END AS MaxValue

                            -- Conditional AVG
                            , CASE 
                                  WHEN t.name IN (''int'',''bigint'',''smallint'',''tinyint'',''decimal'',''numeric'',''float'',''real'',''money'',''smallmoney'')
                                  THEN AVG(CAST(T.' + QUOTENAME(c.name) + ' AS float))
                              END AS AvgValue

                            -- Conditional Max Length
                            , CASE 
                                  WHEN t.name IN (''varchar'',''nvarchar'',''char'',''nchar'',''text'',''ntext'')
                                  THEN MAX(LEN(T.' + QUOTENAME(c.name) + '))
                              END AS MaxLength
                        FROM ' + @FromClause + ' AS T2
                    ) AS Stats
                ) X;
                '
            FROM sys.columns c
            JOIN sys.types t ON c.user_type_id = t.user_type_id
            WHERE c.object_id = OBJECT_ID(QUOTENAME(@SchemaName) + '.' + QUOTENAME(@ObjectName))
        ) AS Q;

        --------------------------------------------------------------------------
        -- 5. Execute profiling SQL
        --------------------------------------------------------------------------
        EXEC sp_executesql @DynamicSQL;

        --------------------------------------------------------------------------
        -- 6. Return resultset R
        --------------------------------------------------------------------------
        SELECT *
        FROM #ProfileResults
        ORDER BY ColumnName;

    END TRY
    BEGIN CATCH
        DECLARE @ErrMessage NVARCHAR(2048) = ERROR_MESSAGE();
        RAISERROR(@ErrMessage, 16, 1);
    END CATCH
END
GO
